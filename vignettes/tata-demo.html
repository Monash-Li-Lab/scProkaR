Simulate a center-rooted multi-drug trajectory

We start with a simulated dataset where the earliest cells sit near the middle of the embedding. This makes the geometry intentionally awkward for topology-only trajectory methods:

  • drug_A extends in one direction and emerges early, around 24 hours
  • drug_C extends along a third arm and emerges around 48 hours
  • drug_B extends in the opposite direction to drug_A and emerges late, around 96 hours

The idea is to create a case where time should help orient the abstract graph, because the root is not sitting at an obvious visual tip.

set.seed(101)

sce <- simulate_tata_multidrug_sce(
  n_cells = 5000,
  n_features = 100,

benchmark with other tools

This section compares the time-aware TATA workflow against two widely used alternatives on the same simulated dataset:

  • slingshot, a branching trajectory method based on cluster topology and principal curves
  • scShaper, a fast ensemble method designed for linear pseudotime inference

The benchmark is intentionally not summarized by a single “winner” score, because this simulation was designed to stress a specific weakness of topology-only methods: the biological origin sits near the middle of the embedding, and one treatment branch emerges later than the others. We therefore look at several complementary metrics:

  • graph_based_trustworthiness: an embedding-based score that asks whether cells that become neighbours in the inferred trajectory space were also close in the known latent trajectory
  • neighborhood_preservation_score: an embedding-based score that measures how much the local k-nearest-neighbour structure is preserved between the known latent trajectory and the inferred trajectory space
  • wasserstein_distance_for_embedding: a higher-is-better score derived from a Wasserstein-style comparison between the pairwise-distance distributions of the known latent trajectory and the inferred trajectory space
  • local_time_concordance: the fraction of local cell-cell graph edges whose inferred pseudotime direction agrees with the true experimental time direction
  • local_temporal_direction_score: a higher-is-better transformation of the local inversion rate, so values near 1 indicate that local pseudotime directions mostly agree with experimental time
  • adjacent_timepoint_auc: for each pair of adjacent timepoints, the probability that later cells have larger inferred pseudotime than earlier cells
  • branch_onset_order_score: a higher-is-better rescaling of the Kendall correlation between the known branch-emergence order (drug_A < drug_C < drug_B) and the inferred onset order
  • branch_onset_accuracy: a higher-is-better transformation of the normalized branch-onset error

Because scShaper is not distributed through Bioconductor, the benchmark code below is optional and will be skipped automatically if either external package is unavailable. The vignette still renders without the benchmark, but when the packages are installed this section provides a useful side-by-side comparison.

Benchmark helpers

The next chunk defines a few small helper functions used only in this vignette. The more generally useful temporal benchmark metrics are now available directly from the package; the local helpers below are mainly for constructing the paper-inspired latent spaces and computing the embedding-based benchmark metrics.

if (have_benchmark_pkgs) {
  scale_to_unit <- function(x) {
    rng <- range(x, na.rm = TRUE)
    if (!all(is.finite(rng)) || diff(rng) < .Machine$double.eps) {
      return(rep(0, length(x)))
    }
    (x - rng[1]) / diff(rng)
  }

  normalize_rows <- function(mat) {
    rs <- rowSums(mat, na.rm = TRUE)
    rs[!is.finite(rs) | rs <= 0] <- 1
    sweep(mat, 1, rs, "/")
  }

  stratified_sample_idx <- function(sce, n_per_group = 70, seed = 101) {
    set.seed(seed)
    group <- interaction(sce$condition, sce$timepoint, drop = TRUE)
    split_idx <- split(seq_len(ncol(sce)), group)
    unlist(lapply(split_idx, function(idx) {
      sample(idx, min(length(idx), n_per_group))
    }), use.names = FALSE)
  }

  make_truth_space <- function(sce) {
    pt <- scale_to_unit(sce$progress)
    branch_levels <- c("drug_A", "drug_B", "drug_C")
    branch_mat <- sapply(branch_levels, function(branch_name) {
      pt * ifelse(sce$simulated_branch == branch_name, sce$branch_activation, 0)
    })
    colnames(branch_mat) <- paste0("truth_", branch_levels)
    cbind(progress = pt, branch_mat)
  }

  make_tata_space <- function(tata_result, sce) {
    cluster_col <- tata_result$parameters$cluster_col
    graph <- tata_result$cluster_graph
    cluster_ids <- as.character(SummarizedExperiment::colData(sce)[[cluster_col]])
    cell_pt <- orient_pseudotime_to_time(
      SummarizedExperiment::colData(sce)[["tata_pseudotime_scaled"]],
      sce$timepoint
    )

    terminals <- igraph::V(graph)$name[igraph::degree(graph, mode = "out") == 0]
    if (length(terminals) == 0L) {
      terminals <- tail(
        tata_result$cluster_pseudotime$cluster[
          order(tata_result$cluster_pseudotime$pseudotime)
        ],
        1
      )
    }

    terminal_weights <- matrix(
      0,
      nrow = igraph::vcount(graph),
      ncol = length(terminals),
      dimnames = list(igraph::V(graph)$name, paste0("tata_", terminals))
    )

    for (cluster_name in rownames(terminal_weights)) {
      dist_out <- as.numeric(igraph::distances(
        graph,
        v = cluster_name,
        to = terminals,
        mode = "out"
      ))
      reachable <- is.finite(dist_out)

      if (any(reachable)) {
        terminal_weights[cluster_name, reachable] <- 1
      } else {
        dist_undirected <- as.numeric(igraph::distances(
          igraph::as_undirected(graph, mode = "collapse"),
          v = cluster_name,
          to = terminals,
          mode = "all"
        ))
        closest <- which.min(dist_undirected)
        terminal_weights[cluster_name, closest] <- 1
      }
    }

    terminal_weights <- normalize_rows(terminal_weights)
    cell_weights <- terminal_weights[cluster_ids, , drop = FALSE]
    cell_weights[!is.finite(cell_weights)] <- 0
    branch_space <- sweep(cell_weights, 1, cell_pt, "*")
    cbind(pseudotime = cell_pt, branch_space)
  }

  make_slingshot_space <- function(sce_sling, truth_reference) {
    pt <- slingshot::slingPseudotime(sce_sling)
    wt <- slingshot::slingCurveWeights(sce_sling)

    if (is.null(dim(pt))) {
      pt <- matrix(pt, ncol = 1)
    }
    if (is.null(dim(wt))) {
      wt <- matrix(wt, ncol = 1)
    }

    pt_scaled <- pt
    for (j in seq_len(ncol(pt_scaled))) {
      pt_scaled[, j] <- orient_pseudotime_to_time(pt_scaled[, j], truth_reference)
    }

    wt[!is.finite(wt)] <- 0
    pt_scaled[!is.finite(pt_scaled)] <- 0
    wt <- normalize_rows(wt)

    aggregate_pt <- rowSums(pt_scaled * wt)
    aggregate_pt <- orient_pseudotime_to_time(aggregate_pt, truth_reference)

    branch_space <- pt_scaled * wt
    colnames(branch_space) <- paste0("sling_lineage_", seq_len(ncol(branch_space)))
    cbind(pseudotime = aggregate_pt, branch_space)
  }

  make_scshaper_space <- function(sce_scshaper, truth_reference) {
    sc_meta <- S4Vectors::metadata(sce_scshaper)[["scshaper"]]
    pt <- sc_meta[["continuous.pseudotime"]]
    if (any(!is.finite(pt))) {
      pt <- sc_meta[["average.discrete.pseudotime"]]
    }
    pt <- orient_pseudotime_to_time(pt, truth_reference)
    cbind(pseudotime = pt)
  }

  graph_based_trustworthiness <- function(truth_space, inferred_space, idx, k = 15L) {
    truth_mat <- as.matrix(truth_space[idx, , drop = FALSE])
    inferred_mat <- as.matrix(inferred_space[idx, , drop = FALSE])
    n <- nrow(truth_mat)

    if (n < 4L) {
      return(NA_real_)
    }

    k <- max(1L, min(as.integer(k), n - 2L))

    truth_dist <- as.matrix(stats::dist(truth_mat))
    inferred_dist <- as.matrix(stats::dist(inferred_mat))
    diag(truth_dist) <- Inf
    diag(inferred_dist) <- Inf

    truth_knn <- t(apply(truth_dist, 1, order))[ , seq_len(k), drop = FALSE]
    inferred_knn <- t(apply(inferred_dist, 1, order))[ , seq_len(k), drop = FALSE]
    truth_rank <- t(apply(truth_dist, 1, rank, ties.method = "average"))

    penalty <- 0
    for (i in seq_len(n)) {
      false_neighbors <- setdiff(inferred_knn[i, ], truth_knn[i, ])
      if (length(false_neighbors) > 0L) {
        penalty <- penalty + sum(truth_rank[i, false_neighbors] - k)
      }
    }

    denom <- n * k * (2 * n - 3 * k - 1)
    if (denom <= 0) {
      return(NA_real_)
    }

    1 - (2 * penalty) / denom
  }

  neighborhood_preservation_score <- function(truth_space, inferred_space, idx, k = 15L) {
    truth_mat <- as.matrix(truth_space[idx, , drop = FALSE])
    inferred_mat <- as.matrix(inferred_space[idx, , drop = FALSE])
    n <- nrow(truth_mat)

    if (n < 3L) {
      return(NA_real_)
    }

    k <- max(1L, min(as.integer(k), n - 1L))

    truth_dist <- as.matrix(stats::dist(truth_mat))
    inferred_dist <- as.matrix(stats::dist(inferred_mat))
    diag(truth_dist) <- Inf
    diag(inferred_dist) <- Inf

    truth_knn <- t(apply(truth_dist, 1, order))[ , seq_len(k), drop = FALSE]
    inferred_knn <- t(apply(inferred_dist, 1, order))[ , seq_len(k), drop = FALSE]

    mean(vapply(seq_len(n), function(i) {
      length(intersect(truth_knn[i, ], inferred_knn[i, ])) / k
    }, numeric(1)))
  }

  wasserstein_distance_for_embedding <- function(
      truth_space,
      inferred_space,
      idx,
      n_quantiles = 200L) {
    truth_dist <- scale_to_unit(as.numeric(stats::dist(truth_space[idx, , drop = FALSE])))
    inferred_dist <- scale_to_unit(as.numeric(stats::dist(inferred_space[idx, , drop = FALSE])))

    if (length(truth_dist) < 3L || length(inferred_dist) < 3L) {
      return(NA_real_)
    }

    probs <- seq(0, 1, length.out = max(20L, as.integer(n_quantiles)))
    truth_q <- stats::quantile(truth_dist, probs = probs, type = 8, na.rm = TRUE)
    inferred_q <- stats::quantile(inferred_dist, probs = probs, type = 8, na.rm = TRUE)
    w1 <- mean(abs(truth_q - inferred_q), na.rm = TRUE)

    1 / (1 + w1)
  }
}
